## Converter 模板

```java
package com.wms.xxx.converter;

import com.wms.xxx.domain.entity.WmsXxx;
import com.wms.xxx.domain.vo.XxxVo;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Component
@RequiredArgsConstructor
public class XxxConverter {

    private final YyyMapper yyyMapper;

    public XxxVo toVo(WmsXxx entity, Map<Long, Yyy> yyyMap) {
        XxxVo vo = new XxxVo();
        vo.setId(entity.getId());
        // ... 字段映射
        if (entity.getYyyId() != null) {
            Yyy yyy = yyyMap.get(entity.getYyyId());
            if (yyy == null) {
                yyy = yyyMapper.selectById(entity.getYyyId());
            }
            if (yyy != null) {
                vo.setYyyName(yyy.getName());
            }
        }
        return vo;
    }

    public List<XxxVo> toVoList(List<WmsXxx> entities, Map<Long, Yyy> yyyMap) {
        return entities.stream()
                .map(e -> toVo(e, yyyMap))
                .collect(Collectors.toList());
    }
}
```
